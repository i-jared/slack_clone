CREATE OR REPLACE FUNCTION "public"."authorize"("requested_permission" "public"."app_permission", "user_id" "uuid") RETURNS boolean
    LANGUAGE "plpgsql" SECURITY DEFINER
    AS $$
  declare
    bind_permissions int;
  begin
    select
      count(*)
    from public.role_permissions
    inner join public.user_roles on role_permissions.role = user_roles.role
    where
      role_permissions.permission = authorize.requested_permission and
      user_roles.user_id = authorize.user_id
    into bind_permissions;

    return bind_permissions > 0;
  end;
$$;


ALTER FUNCTION "public"."authorize"("requested_permission" "public"."app_permission", "user_id" "uuid") OWNER TO "postgres";


CREATE OR REPLACE FUNCTION "public"."handle_new_user"() RETURNS "trigger"
    LANGUAGE "plpgsql" SECURITY DEFINER
    SET "search_path" TO ''
    AS $$
  declare is_admin boolean;
  begin
    insert into public.users (id, username)
    values (new.id, new.email);

    select count(*) = 1 from auth.users into is_admin;

    if position('+supaadmin@' in new.email) > 0 then
      insert into public.user_roles (user_id, role) values (new.id, 'admin');
    elsif position('+supamod@' in new.email) > 0 then
      insert into public.user_roles (user_id, role) values (new.id, 'moderator');
    end if;

    return new;
  end;
$$;

ON storage.objects FOR SELECT
USING (bucket_id = 'avatars');

TO authenticated
WITH CHECK (
    bucket_id = 'avatars'

    AND (
        lower(right(name, 4)) IN ('.jpg', '.png', '.gif')
        OR lower(right(name, 5)) = '.jpeg'
    )
);

- direct_messages ↔ users: Many-to-one relationships through sender_id and recipient_id
- messages ↔ users: Many-to-one through user_id
- messages ↔ channels: Many-to-one through channel_id
- messages ↔ messages: Self-referential through parent_id (for thread replies)
- channels ↔ users: Many-to-one through created_by

2. Common Data Patterns:
- Timestamps: All tables use timestamptz for inserted_at/updated_at
- User References: Always UUID type, referencing auth.users
- Message Content: Always TEXT type with no length limit
- Attachments: Always JSONB type for flexible metadata storage
- Slugs: Used for channel identifiers, TEXT type

3. Security Patterns:
- RLS Policies: All tables have row-level security
- Authentication: Uses Supabase auth.users as source of truth
- File Access: Controlled by user_id in storage path

4. Performance Considerations:
- Indexes exist on all foreign key columns
- Composite indexes on frequently queried combinations
- JSONB for flexible metadata without schema changes

5. Business Logic Rules:
- Messages can be threaded (parent_id)
- Users can have multiple direct message conversations
- Channels are created by users but accessible to all
- File attachments are scoped to user folders

6. Type Definitions:
- user_status: ENUM ('ONLINE', 'OFFLINE')
- app_role: ENUM ('admin', 'moderator')
- app_permission: ENUM ('channels.delete', 'messages.delete')

7. Common Queries:
- Get user's direct messages:
  SELECT * FROM direct_messages 
  WHERE sender_id = auth.uid() OR recipient_id = auth.uid()
  
- Get channel messages:
  SELECT * FROM messages 
  WHERE channel_id = :channel_id
  
- Get threaded replies:
  SELECT * FROM messages 
  WHERE parent_id = :message_id

8. Storage Patterns:
- Avatar path format: avatars/{user_id}/{timestamp}-{random}.{ext}
- Attachment path format: message_attachments/{user_id}/{timestamp}-{random}.{ext}
- File size limits: 5MB per file
- Supported formats: jpg, jpeg, png, gif, pdf (attachments only)

9. Error Handling:
- Foreign key violations indicate invalid references
- Check constraints enforce enum values
- RLS policies may cause permission denied errors
- Storage policies may reject invalid file types

10. Development Notes:
- Local development uses Supabase CLI
- Migrations are timestamp prefixed
- RLS policies should be tested with different user roles
- Storage buckets require authenticated user context

11. Common Edge Cases:
- Deleting users requires cascade handling
- Message edits maintain updated_at timestamp
- File uploads must validate type and size
- Direct messages need bidirectional privacy

12. Authentication Flow:
- Users table auto-populated from auth.users
- Status updates require manual intervention
- Avatar updates trigger storage events
- Session handling through Supabase client

13. Optimization Opportunities:
- Materialized views for frequent queries
- Partial indexes for specific conditions
- Vacuum scheduling for maintenance
- Connection pooling configuration

14. Debugging Tips:
- Check RLS policies first for access issues
- Verify user authentication state
- Confirm storage bucket existence
- Validate foreign key relationships
